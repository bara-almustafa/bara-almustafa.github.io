# كسر حماية كلمة المرور 
وبعد لما اخذنا كلمات المرور الgrafana  بخزن الhashes في Database  من نوع  PBKDF2-HMAC-SHA256
\
فمنحتاج كود برمجي بلغة بايثون عشان نحوله لصيغة نكرس حمايتها بhashcat 
 
```python
#!/usr/bin/env python3
"""Format a stored SHA-256 hash + salt into hashcat's sha256:iter:salt:hash form."""

import base64
import binascii
import sys

PASSWORD_HEX = (
    "dc6becccbb57d34daf4a4e391d2015d3350c60df3608e9e99b5291e47f3e5cd3"
    "9d156be220745be3cbe49353e35f53b51da8"
)
SALT_STR = "LCBhdtJWjl"
ITERATIONS = 10000


def b64(data: bytes) -> str:
    """Base64-encode raw bytes as an ASCII string."""
    return base64.b64encode(data).decode("utf-8")


def build_hashcat_line(password_hex: str, salt: str, iterations: int) -> str:
    try:
        raw = binascii.unhexlify(password_hex)
    except (binascii.Error, ValueError) as exc:
        sys.exit(f"ERROR: PASSWORD_HEX is not valid hex: {exc}")

    return f"sha256:{iterations}:{b64(salt.encode('utf-8'))}:{b64(raw)}"


def main() -> None:
    print(build_hashcat_line(PASSWORD_HEX, SALT_STR, ITERATIONS))


if __name__ == "__main__":
    main()
```
